        'Avatar',
        'Interstellar',
        'The Dark Knight',
        'Avengers Endgame',
        'Doctor Strange',
        'Dune',
        'The Matrix',
        'Gravity'
    ],

    'Genre': [
        'Sci-Fi',
        'Romance',
        'Sci-Fi',
        'Sci-Fi',
        'Action',
        'Action',
        'Sci-Fi',
        'Sci-Fi',
        'Sci-Fi',
        'Sci-Fi'
    ],

    'Rating': [
        8.8,
        7.9,
        7.8,
        8.6,
        9.0,
        8.4,
        7.5,
        8.2,
        8.7,
        7.7
    ],

    'Year': [
        2010,
        1997,
        2009,
        2014,
        2008,
        2019,
        2016,
        2021,
        1999,
        2013
    ]
}

# =========================================================
# CREATING DATAFRAME
# =========================================================

movies = pd.DataFrame(movie_data)

# =========================================================
# DISPLAYING DATASET
# =========================================================

print("\n=============== MOVIE DATASET ===============\n")
print(movies)

# =========================================================
# DATASET INFORMATION
# =========================================================

print("\n=============== DATASET INFORMATION ===============\n")
print(movies.info())

# =========================================================
# NULL VALUES
# =========================================================

print("\n=============== NULL VALUES ===============\n")
print(movies.isnull().sum())

# =========================================================
# TOP RATED MOVIES
# =========================================================

print("\n=============== TOP RATED MOVIES ===============\n")

sorted_movies = movies.sort_values(by='Rating', ascending=False)

print(sorted_movies)

# =========================================================
# SEARCH MOVIE FUNCTION
# =========================================================

def search_movie(movie_name):

    found = movies[movies['Movie_Name'].str.lower() == movie_name.lower()]

    if len(found) > 0:
        print("\nMovie Found Successfully!\n")
        print(found)

    else:
        print("\nMovie Not Found!\n")

# =========================================================
# RECOMMEND MOVIES FUNCTION
# =========================================================

def recommend_movie(movie_name):

    movie_name = movie_name.lower()

    selected_movie = movies[
        movies['Movie_Name'].str.lower() == movie_name
    ]

    if len(selected_movie) == 0:
        print("\nMovie Not Found In Database!\n")
        return

    genre = selected_movie.iloc[0]['Genre']

    print("\nSelected Movie Genre:", genre)

    recommendations = movies[
        movies['Genre'] == genre
    ]

    print("\n=========== RECOMMENDED MOVIES ===========\n")

    for i in range(len(recommendations)):

        print(
            i + 1,
            ".",
            recommendations.iloc[i]['Movie_Name'],
            " Rating:",
            recommendations.iloc[i]['Rating']
        )

# =========================================================
# AVERAGE RATING
# =========================================================

average_rating = np.mean(movies['Rating'])

print("\nAverage Movie Rating:", average_rating)

# =========================================================
# HIGHEST RATED MOVIE
# =========================================================

highest = movies[movies['Rating'] == movies['Rating'].max()]

print("\n=============== HIGHEST RATED MOVIE ===============\n")

print(highest)

# =========================================================
# LOWEST RATED MOVIE
# =========================================================

lowest = movies[movies['Rating'] == movies['Rating'].min()]

print("\n=============== LOWEST RATED MOVIE ===============\n")

print(lowest)

print("\n=============== GENRE COUNTS ===============\n")

# =========================================================

# =========================================================

plt.figure(figsize=(10, 5))

plt.bar(

# =========================================================

# =========================================================
# PIE CHART
# =========================================================

genre_counts = movies['Genre'].value_counts()

plt.figure(figsize=(7, 7))

    genre_counts,

    autopct='%1.1f%%'

plt.show()
